"""Offline r1 sample binding and audit validation, without corpus publication."""
from collections import Counter
from dataclasses import asdict
import hashlib
import json
from pathlib import Path
from types import SimpleNamespace

from nl2sparql.dataset.bilingual.assembly import (
    AssemblyValidationError, _expanded_record, assign_group_splits,
    select_audit_sample, validate_audit_evidence, validate_stage_a_source,
)
from nl2sparql.dataset.bilingual.contracts import AuditEvent, SplitConfig, load_catalog
from nl2sparql.dataset.bilingual.rendering import (
    RENDERER_VERSION, expanded_record_digest, normalize_question, render_pattern,
)
from nl2sparql.dataset.templates import load_templates

sample_path=Path('docs/evidence/bilingual-training-audit-r1-sample-2026-10-15.jsonl')
audit_path=Path('docs/evidence/bilingual-training-audit-r1-decisions-2026-10-15.jsonl')
source_path=Path('data/dataset/raw/synthetic-stage-a-v2.jsonl')
manifest_path=Path('data/dataset/raw/generation-config-v2.json')
catalog_path=Path('src/nl2sparql/dataset/bilingual/templates.json')
source_bytes=source_path.read_bytes()
validate_stage_a_source(source_bytes,manifest_path.read_bytes(),require_accepted=True)
sources={row['id']:row for row in map(json.loads,source_bytes.splitlines())}
templates=load_templates(); template_index={t['id']:t for t in templates}
entries={e.id:e for e in load_catalog(catalog_path,templates).entries}
documents=[json.loads(line) for line in sample_path.read_text().splitlines()]
records=tuple(_expanded_record(row) for row in documents)
events=tuple(AuditEvent(**json.loads(line)) for line in audit_path.read_text().splitlines())
assert len(records)==len(events)==200
for record in records:
    source=sources[record.source_record_id]; entry=entries[record.catalog_entry_id]
    template=template_index[source['template_id']]
    assert record.id==f'{source["id"]}__{entry.language}__{entry.style}'
    assert record.language==entry.language and record.style==entry.style
    assert record.semantic_family_id==source['id']
    assert entry.template_id==source['template_id']
    assert record.catalog_entry_sha256==entry.content_sha256
    assert record.source_record_sha256==source['record_sha256']
    assert record.source_template_sha256==source['template_sha256']
    assert dict(record.slot_values)==dict(record.semantic_anchors)==source['slot_values']
    assert record.sql==source['sql']
    assert record.expected_columns==tuple(template['expected_columns'])
    assert record.schema_elements==tuple(source['schema_elements'])
    assert [dict(entity) for entity in record.entities_used]==source['entities_used']
    assert record.question==render_pattern(entry,source['slot_values'])
    assert record.normalized_question==normalize_question(record.question)
    assert record.renderer_version==RENDERER_VERSION
    assert (record.author_type,record.review_type,record.producer_type)==('agent','agent-reviewed','deterministic_template_renderer')
    assert record.provider is record.generation_model is None
    assert record.api_request_count==record.recorded_cost_usd==0
    assert record.record_sha256==expanded_record_digest(record)
assert assign_group_splits(records,SplitConfig(seed=42,development_percent=10))==tuple(sorted(records,key=lambda r:r.id))
# Sample selection depends on identities only. Project the entire real source
# population, not synthetic evidence, to verify selection without a second
# expensive full text/diversity expansion (CLI20 already passed those gates).
population_identities=tuple(SimpleNamespace(id=f'{source["id"]}__{entry.language}__{entry.style}',
    language=entry.language,style=entry.style,catalog_entry_id=entry.id)
    for source in sources.values() for entry in entries.values() if entry.template_id==source['template_id'])
assert len(population_identities)==8000
for language in ('en','vi'):
    expected=select_audit_sample(population_identities,language,seed=42)
    assert {r.id for r in records if r.language==language}==set(expected)
    strata=Counter(r.catalog_entry_id for r in records if r.language==language)
    assert len(strata)==100 and set(strata.values())=={1}
summaries=validate_audit_evidence(events,records=records,seed=42)
assert all(e.decision==('accept' if e.faithful and e.natural else 'reject') for e in events)
old_events=tuple(AuditEvent(**json.loads(line)) for line in Path('docs/evidence/bilingual-training-audit-decisions-2026-10-15.jsonl').read_text().splitlines())
try:
    validate_audit_evidence(old_events,records=records,seed=42)
except AssemblyValidationError as error:
    assert 'record digest does not match' in str(error)
else:
    raise AssertionError('Historical decisions must reject the new catalogue records')
assert not Path('data/dataset/processed/bilingual-training.jsonl').exists()
assert not Path('data/dataset/processed/bilingual-training-manifest.json').exists()
print(json.dumps({'status':'passed','revision':'audit-correction-r1','reviewer_type':'agent',
    'independent_or_human_review':False,'api_request_count':0,'canonical_publication':False,
    'binding_validation':'passed','full_population_sample_selection':'passed',
    'historical_decisions_rejected_for_new_records':True,
    'thresholds':{'faithful_minimum_per_100':95,'natural_minimum_per_100':90},
    'languages':[asdict(summary) for summary in summaries],
    'source_and_evidence_sha256':{str(path):hashlib.sha256(path.read_bytes()).hexdigest()
        for path in (source_path,manifest_path,catalog_path,sample_path,audit_path)}},indent=2))
