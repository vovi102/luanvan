"""Execute the approved exact phase-A scope once; never submit live jobs."""
import hashlib
import json
import os
import time
from collections import Counter, defaultdict
from datetime import UTC, datetime
from pathlib import Path

import google.auth
from google.auth.transport.requests import AuthorizedSession
from google.cloud import bigquery

ROOT = Path.cwd()
OUT = ROOT / 'docs/evidence/t54-phase-a-2026-10-24'
SCOPE_PATH = ROOT / 'docs/evidence/t54-bigquery-review-2026-10-24/dry-run-scope.json'
DECISION_PATH = OUT / 'decision.json'

def sha(data):
    return hashlib.sha256(data).hexdigest()
def now():
    return datetime.now(UTC).isoformat()
def dump(path, obj):
    with path.open('x') as file:
        json.dump(obj, file, ensure_ascii=False, indent=2, allow_nan=False)
        file.write('\n')
        file.flush()
        os.fsync(file.fileno())
def read(path):
    return json.loads(path.read_bytes())

decision = read(DECISION_PATH)
scope = read(SCOPE_PATH)
assert decision['status'] == 'authorized_dry_run_only'
assert decision['authorization_recorded'] is True
assert decision['requested_scope']['live_execution'] is False
assert sha(SCOPE_PATH.read_bytes()) == decision['dry_run_scope_sha256']
assert sha((ROOT / decision['proposal_path']).read_bytes()) == decision['proposal_file_sha256']
assert sha((ROOT / decision['privacy_policy_path']).read_bytes()) == decision['privacy_policy_sha256']
assert scope['dry_run'] is True and scope['allow_live_execution'] is False
assert scope['project'] == 'nl2sparql-thesis' and scope['location'] == 'US'
assert scope['unique_sql_count'] == len(scope['queries']) == 143
assert read(OUT / 'billing-refresh.json')['billingEnabled'] is False
assert read(OUT / 'privacy-bound-runs.json')['runs'].__len__() == 12
assert not (OUT / 'dry-run-journal.jsonl').exists(), 'No resume/retry permitted'
for query in scope['queries']:
    assert sha(query['sql'].encode()) == query['sql_sha256']
config = bigquery.QueryJobConfig(dry_run=True, use_legacy_sql=False, use_query_cache=False, maximum_bytes_billed=scope['maximum_bytes_billed'])
config_json = config.to_api_repr()
assert config_json['dryRun'] is True
assert config_json['query']['useLegacySql'] is False
assert config_json['query']['useQueryCache'] is False
assert int(config_json['query']['maximumBytesBilled']) == 24 * 1024**3
credentials, _ = google.auth.default(scopes=['https://www.googleapis.com/auth/cloud-platform'])
session = AuthorizedSession(credentials, max_refresh_attempts=0)
client = bigquery.Client(project=scope['project'], credentials=credentials, _http=session, location=scope['location'])
dump(OUT / 'request-policy.json', {'created_at_utc':now(), 'decision_file_sha256':sha(DECISION_PATH.read_bytes()), 'scope_file_sha256':sha(SCOPE_PATH.read_bytes()), 'client_version':bigquery.__version__, 'job_config':config_json, 'project':scope['project'], 'location':scope['location'], 'http_retry':None, 'job_retry':None, 'auth_max_refresh_attempts':0, 'dry_run_requests_max':143, 'result_api_calls':0, 'automatic_resume':False})
records = []
previous = '0'*64
with (OUT / 'dry-run-journal.jsonl').open('x') as journal:
    def append(payload):
        global previous
        base = {'index':len(records), 'previous_sha256':previous, 'payload':payload}
        digest = sha(json.dumps(base,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode())
        journal.write(json.dumps(base | {'record_sha256':digest},ensure_ascii=False)+'\n')
        journal.flush()
        os.fsync(journal.fileno())
        previous = digest
    for index, query in enumerate(scope['queries']):
        intent = {'kind':'dry_run_intent','scope_index':index,'sql_sha256':query['sql_sha256'],'at_utc':now(),'dry_run':True,'project':scope['project'],'location':scope['location']}
        append(intent)
        started = time.monotonic()
        record = {'scope_index':index,'sql_sha256':query['sql_sha256'],'started_at_utc':intent['at_utc'],'client_version':bigquery.__version__,'dry_run':True,'logical_use_counts':dict(Counter(use['role'] for use in query['uses']))}
        try:
            job = client.query(query['sql'], job_config=config, project=scope['project'], location=scope['location'], job_id='t54phasea20261024_' + query['sql_sha256'][:40], retry=None, job_retry=None, timeout=120, api_method=bigquery.enums.QueryApiMethod.INSERT)
            api = job.to_api_repr()
            assert api['configuration']['dryRun'] is True, 'Unexpected live response'
            record.update(response=api, estimated_bytes=job.total_bytes_processed, status='ok')
            if job.errors:
                record.update(status='dry_run_error', errors=job.errors)
            elif not isinstance(job.total_bytes_processed,int) or isinstance(job.total_bytes_processed,bool) or job.total_bytes_processed < 0:
                record.update(status='unknown_estimate')
            elif job.total_bytes_processed > scope['maximum_bytes_billed']:
                record.update(status='estimate_above_24_GiB')
            elif job.total_bytes_processed == 0:
                record.update(status='zero_estimate_requires_review')
        except Exception as exc:
            record.update(status='dry_run_error', estimated_bytes=None, exception_type=type(exc).__name__, error_message=str(exc), provider_errors=getattr(exc,'errors',None))
        record.update(finished_at_utc=now(),elapsed_seconds=time.monotonic()-started)
        append({'kind':'dry_run_outcome',**record})
        records.append(record)
        print(json.dumps({'attempted':len(records),'status':record['status'],'estimated_bytes':record['estimated_bytes'],'sql_sha256':query['sql_sha256'][:12]}),flush=True)
        if record['status'] not in ('ok','zero_estimate_requires_review'):
            break
    append({'kind':'terminal_seal','at_utc':now(),'attempted':len(records),'outcome_counts':dict(Counter(r['status'] for r in records)),'live_requests':0,'unattempted':143-len(records)})
lookup = {record['sql_sha256']:record for record in records}
per_run = defaultdict(lambda: {'expected_queries':0,'estimated_queries':0,'unestimated_queries':0,'known_estimated_bytes':0})
for query in scope['queries']:
    record = lookup.get(query['sql_sha256'])
    for use in query['uses']:
        totals = per_run[use['run_id']]
        totals['expected_queries'] += 1
        if record is not None and record.get('estimated_bytes') is not None:
            totals['estimated_queries'] += 1
            totals['known_estimated_bytes'] += record['estimated_bytes']
        else:
            totals['unestimated_queries'] += 1
for totals in per_run.values():
    totals['complete_estimated_bytes'] = totals['known_estimated_bytes'] if totals['unestimated_queries']==0 else None
summary = {'kind':'phase_A_dry_run_estimates_not_execution_evidence','finished_at_utc':now(),'attempted_requests':len(records),'max_authorized_requests':143,'status_counts':dict(Counter(r['status'] for r in records)),'unattempted_sql_sha256':[query['sql_sha256'] for query in scope['queries'] if query['sql_sha256'] not in lookup], 'stop_reason': records[-1]['status'] if len(records)<143 else 'scope_exhausted','unique_known_estimated_bytes':sum(r.get('estimated_bytes') or 0 for r in records),'logical_uses_expected':1329,'logical_uses_estimated':sum(t['estimated_queries'] for t in per_run.values()),'logical_uses_unestimated':sum(t['unestimated_queries'] for t in per_run.values()),'campaign_complete_estimated_bytes':sum(t['known_estimated_bytes'] for t in per_run.values()) if all(t['unestimated_queries']==0 for t in per_run.values()) else None,'per_run':dict(per_run),'zero_or_unknown_estimates_are_not_free_claims':True,'journal_file_sha256':sha((OUT/'dry-run-journal.jsonl').read_bytes()),'journal_terminal_record_sha256':previous,'live_requests':0,'execution_accuracy':None,'free_allowance_remaining_bytes':None}
dump(OUT/'dry-run-summary.json',summary)
print(json.dumps({'summary':str((OUT/'dry-run-summary.json').relative_to(ROOT)), 'attempted':len(records),'stop_reason':summary['stop_reason']}),flush=True)
