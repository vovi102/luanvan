import hashlib
import json
from dataclasses import asdict
from datetime import datetime
from pathlib import Path

from nl2sparql.evaluation.adapters.b12 import B12AdaptRequest, adapt_b12
from nl2sparql.evaluation.artifacts import (
    load_and_verify_artifact,
    publish_immutable,
    serialize_prediction_run,
    serialize_privacy_review,
)
from nl2sparql.evaluation.contracts import PrivacyReview

root = Path.cwd()
out = root / 'docs/evidence/t54-phase-a-2026-10-24'
read = lambda p: json.loads(Path(p).read_bytes())
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
source_decision = root / 'configs/t54/phase-a-approval-2026-10-24.json'
a = read(source_decision)
for path_key, hash_key in [('proposal_path','proposal_file_sha256'),('privacy_policy_path','privacy_policy_sha256'),('dry_run_scope_path','dry_run_scope_sha256')]:
    assert sha(root / a[path_key]) == a[hash_key]
a.update(status='authorized_dry_run_only', reviewer_id='conversation-user', reviewed_at_utc='2026-10-24T06:56:11+00:00', identity_preference_text='không cần', identity_preference_source='user_reply_to_reviewer_identity_question', reviewer_identity_semantics='Pseudonymous role identifier for the actual approving user in this conversation; no personal name supplied or independently verified.', note='Actual approval recorded from user. User declines personal reviewer identity. Agent records the decision; no agent is represented as reviewer. No live authorization. Original proposal/templates and offline canonical artifacts preserved.')
publish_immutable(out / 'decision.json', (json.dumps(a,ensure_ascii=False,indent=2)+'\n').encode())
w = read(root / 'configs/t54/privacy-review-drafts.json')
sidecars = {}
for r in w['reviews']:
    review = PrivacyReview(baseline_id=r['baseline_id'], test_set_sha256=r['test_set_sha256'], data_egress=r['data_egress'], provider=r['provider'], policy_sha256=r['policy_sha256'], reviewer_id=a['reviewer_id'], reviewed_at=datetime.fromisoformat(a['reviewed_at_utc']), synthetic=False)
    path = out / 'privacy' / f"{r['baseline_id']}-{r['language']}.json"
    publish_immutable(path, serialize_privacy_review(review))
    load_and_verify_artifact(path)
    sidecars[(r['baseline_id'],r['language'])] = path
matrix = read(root / 'docs/evidence/kaggle-b12-generation-matrix-2026-10-23.json')
proposal = read(root / a['proposal_path'])
old_entries = {r['run_id']: r for r in proposal['execution_scope']['runs']}
new_entries=[]
for row in matrix['rows']:
    source = root / row['source_run_json']
    folder = source.parent
    snapshot = root / 'data/dataset/test' / ('test-100.jsonl' if row['language']=='en' else 'test-100-vi.jsonl')
    fresh = adapt_b12(B12AdaptRequest(snapshot, folder/'predictions.jsonl', folder/'native-log.jsonl', folder/'native-report.json', privacy_review_path=sidecars[(row['baseline'],row['language'])], language=row['language'], source_root=root))
    old_path = root / old_entries[row['run_id']]['canonical_path']
    assert sha(old_path) == old_entries[row['run_id']]['canonical_file_sha256']
    old = load_and_verify_artifact(old_path)
    old_data, new_data = asdict(old), asdict(fresh)
    old_cases, new_cases = old_data.pop('cases'), new_data.pop('cases')
    old_data.pop('source_artifacts'); new_data.pop('source_artifacts')
    assert old_data == new_data, 'non-privacy run fields changed'
    for left,right in zip(old_cases,new_cases,strict=True):
        left.pop('privacy'); right.pop('privacy')
        assert left == right, 'case changed beyond privacy'
    path = out / 'canonical' / f'{fresh.run_id}.json'
    publish_immutable(path, serialize_prediction_run(fresh))
    load_and_verify_artifact(path)
    new_entries.append({'run_id':fresh.run_id,'baseline':row['baseline'],'language':row['language'],'canonical_path':str(path.relative_to(root)),'canonical_file_sha256':sha(path),'original_canonical_path':str(old_path.relative_to(root)),'original_canonical_sha256':sha(old_path),'privacy_path':str(sidecars[(row['baseline'],row['language'])].relative_to(root))})
publish_immutable(out/'privacy-bound-runs.json',(json.dumps({'kind':'privacy_rebind_mapping_not_execution_evidence','decision_path':str((out/'decision.json').relative_to(root)),'runs':new_entries},indent=2)+'\n').encode())
billing = read(root/'.local/t54-phase-a-2026-10-24/billing-refresh.json')
assert billing['billingEnabled'] is False
publish_immutable(out/'billing-refresh.json',(json.dumps(billing,indent=2)+'\n').encode())
print(json.dumps({'sidecars':len(sidecars),'new_canonical_runs':len(new_entries),'non_privacy_fields_unchanged':True,'live_authorized':False}))
