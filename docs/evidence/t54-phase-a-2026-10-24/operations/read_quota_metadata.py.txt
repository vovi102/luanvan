"""Read-only metadata probes; no query jobs or service changes."""
import json
from datetime import UTC, datetime
from pathlib import Path
import google.auth
from google.auth.transport.requests import AuthorizedSession

out=Path('docs/evidence/t54-phase-a-2026-10-24')
credentials,_=google.auth.default(scopes=['https://www.googleapis.com/auth/cloud-platform'])
session=AuthorizedSession(credentials,max_refresh_attempts=0)
def get(url, params=None):
    at=datetime.now(UTC).isoformat()
    response=session.get(url,params=params,timeout=60)
    try:
        body=response.json()
    except ValueError:
        body={'text':response.text[:4000]}
    return {'observed_at_utc':at,'method':'GET','url':url,'status_code':response.status_code,'body':body}
project=get('https://cloudresourcemanager.googleapis.com/v1/projects/nl2sparql-thesis')
with (out/'project-metadata-refresh.json').open('x') as f:
    json.dump(project,f,indent=2);f.write('\n')
number=project['body'].get('projectNumber')
quota_pages=[]
if number is not None:
    url=f'https://serviceusage.googleapis.com/v1/projects/{number}/services/bigquery.googleapis.com/consumerQuotaMetrics'
    token=None
    while True:
        params={'view':'FULL','pageSize':200}
        if token:
            params['pageToken']=token
        page=get(url,params)
        quota_pages.append(page)
        if page['status_code'] != 200:
            break
        token=page['body'].get('nextPageToken')
        if not token:
            break
with (out/'quota-metadata-refresh.json').open('x') as f:
    json.dump({'kind':'read_only_quota_probe_not_balance','project_lookup_status':project['status_code'],'pages':quota_pages,'free_allowance_remaining_bytes':None,'note':'Quota limits and gross job history alone do not establish remaining monthly sandbox allowance. No service enablement or quota/IAM change.'},f,indent=2);f.write('\n')
print(json.dumps({'project_metadata_status':project['status_code'],'quota_page_statuses':[p['status_code'] for p in quota_pages],'verified_free_balance':False}))
